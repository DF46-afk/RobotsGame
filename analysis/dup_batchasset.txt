/* ==========================================================================
 * === ASSET PREPARATION ====================================================
 * Measured facts about the four GLBs (verified against their JSON chunks):
 *   city.glb            2455 meshes, rigid -> batched per material into a few
 *                       big indexed VBOs so the city costs ~100 draw calls.
 *   atlas_hangar_ld.glb skinned mech, 51 joints, authored in CENTIMETRES
 *                       (bbox height 0.29u => 29 m at scale 1) => scale 0.1.
 *   kitsune_harpy_rig   skinned rig, 28 joints, metres (~18 u tall).
 *   kid_war_robots.glb  NO skin: 53 rigid robot parts in one file; we split
 *                       them by XZ footprint cluster and bake every robot
 *                       into static batches reused as enemy "walkers".
 * Every step degrades gracefully: a missing/corrupt file never blocks boot.
 * ==========================================================================*/

/** Fetch with progress + graceful failure (never rejects the boot chain). */
async function fetchGLB(url, label, onBytes) {
  try {
    const res = await fetch(url, { cache: 'default', mode: 'cors' });
    if (!res.ok) throw new Error('HTTP ' + res.status);
    const total = +(res.headers.get('content-length') || 0);
    if (!res.body || !total) {
      const buf = await res.arrayBuffer();
      if (onBytes) onBytes(buf.byteLength, buf.byteLength);
      return buf;
    }
    const reader = res.body.getReader();
    const chunks = []; let got = 0;
    for (;;) {
      const { done, value } = await reader.read();
      if (done) break;
      chunks.push(value); got += value.length;
      if (onBytes) onBytes(got, total);
    }
    const out = new Uint8Array(got); let o = 0;
    for (const c of chunks) { out.set(c, o); o += c.length; }
    return out.buffer;
  } catch (e) {
    logLine(label + ': fetch failed (' + e.message + ') — procedural fallback engaged', true);
    return null;
  }
}

function pushArr(dst, src) { for (let i = 0; i < src.length; i++) dst.push(src[i]); }
function mkIdxBuf(gl, arr) {
  const b = gl.createBuffer(); gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, b);
  gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, arr, gl.STATIC_DRAW); return b;
}

/** Merge many small static primitives into few big indexed batches (one group
 *  per material / doubleSided / vertexColor combination).
 *  @returns {Array} batches [{primLike, mat, tris}]
 */
function batchAsset(gl, asset, filter) {
  const groups = new Map();
  const nodes = asset.nodes;
  for (const prim of asset.prims) {
    if (prim.skinned) continue;
    let world = null;
    if (filter) {
      const ni = nodes.findIndex(n => n.mesh === prim.mesh);
      world = ni >= 0 ? nodes[ni].world : null;
      if (!filter(prim, world)) continue;
    }
    const mat = prim.mat;
    const key = (mat ? mat.idx : -1) + '|' + (mat && mat.doubleSided ? 1 : 0) +
                '|' + (prim.hasColor ? 1 : 0) + '|' + (mat && mat.baseTex >= 0 ? 1 : 0);
    let g = groups.get(key);
    if (!g) { g = { mat, prims: [] }; groups.set(key, g); }
    g.prims.push(prim);
  }
  const batches = [];
  for (const g of groups.values()) {
    const P = [], N = [], U = [], C = [], IDX = [];
    let base = 0, hasC = false;
    for (const src of g.prims) {
      const cpu = src.cpu || src.vbo;
      const p = cpu.POSITION; if (!p) continue;
      pushArr(P, p);
      const nv = p.length / 3;
      if (cpu.NORMAL) pushArr(N, cpu.NORMAL);
      else for (let i = 0; i < nv; i++) N.push(0, 1, 0);
      if (cpu.TEXCOORD_0) pushArr(U, cpu.TEXCOORD_0);
      else for (let i = 0; i < nv; i++) U.push(0.5, 0.5);
      if (cpu.COLOR_0) { hasC = true; pushArr(C, cpu.COLOR_0); }
      if (src.indexed && cpu.IBO) {
        const idx = cpu.IBO;
        for (let i = 0; i < idx.length; i++) IDX.push(idx[i] + base);
      } else {
        for (let i = 0; i < nv; i++) IDX.push(base + i);
      }
      base += nv;
    }
    if (!P.length) continue;
    const mk = (arr) => {
      const b = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, b);
      gl.bufferData(gl.ARRAY_BUFFER, arr, gl.STATIC_DRAW); return b;
    };
    const primLike = {
      mesh: -1, prim: -1, mat: g.mat, count: P.length / 3, indexed: true,
      skinned: false, gpu: {}, hasColor: hasC, name: 'batch-' + batches.length,
      bboxMin: [-1e4, -1e4, -1e4], bboxMax: [1e4, 1e4, 1e4]
    };
    primLike.gpu.POSITION = mk(Float32Array.from(P));
    primLike.gpu.NORMAL = mk(Float32Array.from(N));
    primLike.gpu.TEXCOORD_0 = mk(Float32Array.from(U));
    if (hasC) primLike.gpu.COLOR_0 = mk(Float32Array.from(C));
    primLike.ibo = mkIdxBuf(gl, Uint32Array.from(IDX));
    primLike.iboCount = IDX.length; primLike.iboType = gl.UNSIGNED_INT;
    primLike._vao = null;
    batches.push({ primLike, mat: g.mat, tris: IDX.length / 3 });
  }
  return batches;
}

/** Prepare a skin for GPU skinning. The mesh shader reads matrix j from the
 *  RGBA32F strip texels [j*4 .. j*4+3] addressed as x = b % W, y = b / W with
 *  b = j*4 — i.e. a flat 4·count-texel strip padded to width `texW`. */
function prepSkin(r, asset, skin) {
  const W = Math.max(8, skin.count * 4);
  skin.texW = W;
  skin.jointTexData = new Float32Array(W * 4);   // one row of 4-component texels
  let o = 0;
  for (let j = 0; j < skin.count; j++)           // identity prefill
    for (let c = 0; c < 4; c++)
      for (let k = 0; k < 4; k++) skin.jointTexData[o++] = (c === k) ? 1 : 0;
  skin.texBuf = r.makeFloatTexture(W, 1, skin.jointTexData);
  skin.prims = [];
  for (const prim of asset.prims) {
    if (!prim.skinned) continue;
    prim._vao = r.vaoFor(prim);                   // fixed attrib layout
    skin.prims.push(prim);
  }
  return skin;
}
